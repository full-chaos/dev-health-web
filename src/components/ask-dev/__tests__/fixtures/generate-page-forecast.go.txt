// internal/queryapi/capacityforecast/zz_fixture_export.go
package capacityforecast

import (
	"bytes"
	"encoding/json"
	"reflect"
	"time"

	"github.com/full-chaos/dev-health-ops/internal/queryapi/graphqldate"
)

// Scratch only (CHAOS-4613 web parity fixture), never committed.
type zzRow []any

func (r zzRow) Next() bool   { return true }
func (r zzRow) Err() error   { return nil }
func (r zzRow) Close() error { return nil }

func (r zzRow) Scan(dest ...any) error {
	for i, d := range dest {
		reflect.ValueOf(d).Elem().Set(reflect.ValueOf(r[i]))
	}
	return nil
}

// ZZStoredForecastJSON is one stored capacity_forecasts row through the real
// scanForecastRow: the same row acr's forecast-curve fixture reads.
func ZZStoredForecastJSON() ([]byte, error) {
	computed := time.Date(2026, 11, 2, 4, 0, 0, 0, time.UTC)
	team, scope := "chaos", "linear:CHAOS"
	day := func(y int, m time.Month, d int) *time.Time { t := time.Date(y, m, d, 0, 0, 0, 0, time.UTC); return &t }
	u16 := func(v uint16) *uint16 { return &v }
	row := zzRow{
		"forecast-chaos-0001", computed, &team, &scope, uint32(40),
		(*uint32)(nil), (*time.Time)(nil), day(2026, 11, 17), day(2026, 11, 23), day(2026, 12, 2),
		u16(15), u16(21), u16(30), (*uint32)(nil), (*uint32)(nil), (*uint32)(nil),
		3.25, 1.1, uint16(90),
		uint8(0), uint8(0),
		[]uint16{9, 12, 15, 21, 30, 365}, []uint32{1000, 2500, 3000, 2000, 1000, 500}, []uint32{}, []uint32{},
	}
	model, err := scanForecastRow(row)
	if err != nil {
		return nil, err
	}
	// Dates go out through their GraphQL marshaller, as the server sends them.
	encoded, err := json.Marshal(model)
	if err != nil {
		return nil, err
	}
	var fields map[string]any
	if err := json.Unmarshal(encoded, &fields); err != nil {
		return nil, err
	}
	for name, date := range map[string]*graphqldate.Date{"p50Date": model.P50Date, "p85Date": model.P85Date, "p95Date": model.P95Date} {
		if date == nil {
			continue
		}
		var buffer bytes.Buffer
		date.MarshalGQL(&buffer)
		var value any
		if err := json.Unmarshal(buffer.Bytes(), &value); err != nil {
			return nil, err
		}
		fields[name] = value
	}
	return json.MarshalIndent(fields, "", "    ")
}

// cmd/zzcapacity/main.go
// Scratch only (CHAOS-4613): one stored forecast row through ops' real scanForecastRow.
package main

import (
	"fmt"
	"os"

	"github.com/full-chaos/dev-health-ops/internal/queryapi/capacityforecast"
)

func main() {
	encoded, err := capacityforecast.ZZStoredForecastJSON()
	if err != nil {
		panic(err)
	}
	if err := os.WriteFile(os.Args[1], append(encoded, '\n'), 0o644); err != nil {
		panic(err)
	}
	fmt.Printf("bytes=%d\n", len(encoded))
}
